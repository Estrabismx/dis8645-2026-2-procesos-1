```cpp
#include <SPI.h>
#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>


#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 32


#define OLED_RESET     -1
#define SCREEN_ADDRESS 0x3C
Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, OLED_RESET);


#define POT_PIN A0
#define BUTTON_PIN 2


const char* frases[] = {
  "Huye luna, luna, luna.",
  "Si vienen los gitanos."
};
const int cantidadFrases = sizeof(frases) / sizeof(frases[0]);
int fraseActual = 0;


int16_t textWidth, textHeight;
int minX;


bool botonPresionadoAntes = false; // bandera simple


void actualizarMedidasTexto() {
  int16_t x1, y1;
  display.getTextBounds(frases[fraseActual], 0, 0, &x1, &y1, (uint16_t*)&textWidth, (uint16_t*)&textHeight);
  minX = SCREEN_WIDTH - textWidth;
  if (minX > 0) minX = 0;


  Serial.print("Frase: ");
  Serial.println(frases[fraseActual]);
}


void setup() {
  Serial.begin(9600);
  pinMode(BUTTON_PIN, INPUT_PULLUP);


  delay(500);


  if (!display.begin(SSD1306_SWITCHCAPVCC, SCREEN_ADDRESS)) {
    Serial.println(F("SSD1306 allocation failed"));
    for (;;);
  }


  display.display();
  delay(1000);
  display.clearDisplay();


  display.setTextSize(2);
  display.setTextColor(SSD1306_WHITE);
  display.setTextWrap(false);


  actualizarMedidasTexto();
}


void loop() {
  // Lectura simple del botón
  bool botonPresionadoAhora = (digitalRead(BUTTON_PIN) == LOW);


  // Detecta el momento exacto en que se presiona (no mientras se mantiene)
  if (botonPresionadoAhora && !botonPresionadoAntes) {
    fraseActual = (fraseActual + 1) % cantidadFrases;
    actualizarMedidasTexto();
    delay(200); // pequeño retardo anti-rebote, simple pero efectivo
  }


  botonPresionadoAntes = botonPresionadoAhora;


  // Lógica del potenciómetro
  int potValue = analogRead(POT_PIN);
  int xPos = map(potValue, 0, 1023, 0, minX);
  int yPos = (SCREEN_HEIGHT - textHeight) / 2;


  display.clearDisplay();
  display.setTextWrap(false);
  display.setCursor(xPos, yPos);
  display.print(frases[fraseActual]);
  display.display();


  delay(20);
}
```
